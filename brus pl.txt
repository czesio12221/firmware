// Domyślna pusta funkcja: dla płytek, które definiują HAS_ENCODER, ale nie implementują
// pollEncoder() (nie powinno się zdarzyć, ale pozwala na poprawną pracę konsolidatora).
void __attribute__((weak)) pollEncoder(void) {}

// Dedykowane zadanie o wysokim priorytecie i krótkim cyklu, którego jedynym celem jest próbkowanie
// linii A/B enkodera obrotowego -- odzwierciedla to działanie input_srv z portu Flippera,
// które uruchamia encoder_poll() we własnym wątku co 4ms, niezależnie od pracy GUI/aplikacji,
// dzięki czemu odczyt kwadraturowy nigdy nie jest opóźniany przez renderowanie ekranu ani przez to,
// czy poprzednie zdarzenie wejściowe zostało już przetworzone. Istnieje tylko na płytkach z HAS_ENCODER;
// inne płytki nie ponoszą żadnych kosztów wydajnościowych z tego tytułu.
static void taskEncoderPoll(void *parameter) {
    while (true) {
        pollEncoder();
        vTaskDelay(pdMS_TO_TICKS(4));
    }
}
#endif

TaskHandle_t xHandle;
void __attribute__((weak)) taskInputHandler(void *parameter) {
    auto timer = millis();
    while (true) {
        checkPowerSaveTime();
        // Czasami to zadanie uruchamia się 2 lub więcej razy przed looptask
        // i nawigacja się zawiesza. Pomysł polega na uruchomieniu wykrywania wejścia,
        // jeśli AnyKeyPress ma wartość false, lub ponownym uruchomieniu, jeśli nie zostało odnowione w ciągu 75ms (wartość arbitralna),
        // ponieważ AnyKeyPress będzie mieć wartość true, jeśli nie przeszło przez sprawdzenie (zmienna bool)
        if (!AnyKeyPress || millis() - timer > 75) {
            NextPress = false;
            PrevPress = false;
            UpPress = false;
            DownPress = false;
            SelPress = false;
            EscPress = false;
            AnyKeyPress = false;
            SerialCmdPress = false;
            NextPagePress = false;
            PrevPagePress = false;
            touchPoint.pressed = false;
            touchPoint.Clear();
            checkAndRecoverSysI2CBus();
#ifndef USE_TFT_eSPI_TOUCH
            InputHandler();
#endif
            timer = millis();
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

/*********************************************************************
 **  Funkcja: begin_storage
 **  Ustawienia LittleFS oraz karty SD
 *********************************************************************/
void begin_storage() {
    if (!setupLittleFS()) {
        LittleFS.format();
        setupLittleFS();
    }
    RAM_LOG("po uruchomieniu LittleFS");
    bool checkFS = setupSdCard();
    bruceKonfiguracja.fromFile(checkFS);
    bruceKonfiguracjaPins.fromFile(checkFS);
}
